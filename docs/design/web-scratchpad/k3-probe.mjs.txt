// One-off: a tip that repeats its own words opens only when they are cut short. Vite on 5191.
import { spawn } from "node:child_process";
import { createRequire } from "node:module";
import { join } from "node:path";

const [web] = process.argv.slice(2);
const { chromium } = createRequire(join(web, "package.json"))("@playwright/test");
const vite = spawn(process.execPath, ["node_modules/vite/bin/vite.js", "--port", "5191", "--strictPort"], { cwd: web, stdio: "ignore" });
const say = (k, v) => console.log(`${k}: ${typeof v === "string" ? v : JSON.stringify(v)}`);
try {
  const base = "http://127.0.0.1:5191";
  for (let i = 0; ; i++) {
    try { if ((await fetch(base)).ok) break; } catch {}
    if (i > 100) throw new Error("vite did not start");
    await new Promise((r) => setTimeout(r, 200));
  }
  const browser = await chromium.launch({ channel: "chrome" });
  const page = await browser.newPage({ viewport: { width: 1000, height: 900 } });
  const tip = () => page.evaluate(() => [...document.querySelectorAll("[class*=_tip_]")].filter((e) => e.checkVisibility()).map((e) => e.textContent));
  for (const path of ["/intake", "/compact", "/t/NW-17"]) {
    await page.goto(base + path);
    await page.waitForTimeout(500);
    const cuts = await page.evaluate(() => [...document.querySelectorAll("main [data-tip-cut]")].map((e, i) => {
      const cut = [e, ...e.querySelectorAll("*")].some((x) => x.scrollWidth > x.clientWidth || x.scrollHeight > x.clientHeight);
      e.setAttribute("data-probe", String(i));
      const r = e.getBoundingClientRect();
      return { i, cut, text: e.textContent.slice(0, 30), visible: r.width > 0 && r.top > 0 && r.bottom < innerHeight };
    }).filter((c) => c.visible));
    const one = cuts.find((c) => c.cut);
    const whole = cuts.find((c) => !c.cut);
    for (const [k, c] of [["cut", one], ["whole", whole]]) {
      if (!c) { say(`${path} ${k}`, "none on screen"); continue; }
      await page.mouse.move(1, 1);
      await page.waitForTimeout(300);
      await page.locator(`[data-probe="${c.i}"]`).hover({ force: true });
      await page.waitForTimeout(900);
      say(`${path} ${k} "${c.text}"`, await tip());
    }
    say(`${path} described`, await page.evaluate(() => document.querySelectorAll("main [aria-description]").length));
  }
  await browser.close();
} finally {
  vite.kill();
}
