// One-off: shoot the text area pop-out, closed and open, in both themes. Starts Vite on 5191.
import { spawn } from "node:child_process";
import { createRequire } from "node:module";
import { join } from "node:path";

const [web, out] = process.argv.slice(2);
const { chromium } = createRequire(join(web, "package.json"))("@playwright/test");
const vite = spawn(process.execPath, ["node_modules/vite/bin/vite.js", "--port", "5191", "--strictPort"], { cwd: web, stdio: "ignore" });
try {
  const base = "http://127.0.0.1:5191";
  for (let i = 0; ; i++) {
    try { if ((await fetch(base)).ok) break; } catch {}
    if (i > 100) throw new Error("vite did not start");
    await new Promise((r) => setTimeout(r, 200));
  }
  const browser = await chromium.launch({ channel: "chrome" });
  for (const theme of ["light", "dark"]) {
    const ctx = await browser.newContext({ viewport: { width: 1440, height: 900 }, timezoneId: "Europe/London", locale: "en-GB" });
    await ctx.addInitScript((t) => localStorage.setItem("agentos-theme", t), theme);
    const page = await ctx.newPage();
    const errors = [];
    page.on("pageerror", (e) => errors.push(String(e)));
    await page.goto(`${base}/library/agents/builder`);
    await page.evaluate(() => document.fonts.ready);
    await page.getByRole("textbox", { name: /instructions/i }).first().hover();
    await page.screenshot({ path: join(out, `popout-hover-${theme}.png`) });
    await page.getByRole("button", { name: /whole window/i }).first().click();
    await page.waitForTimeout(300);
    await page.screenshot({ path: join(out, `popout-open-${theme}.png`) });
    await page.keyboard.press("Escape");
    await page.waitForTimeout(200);
    const closed = await page.evaluate(() => [...document.querySelectorAll("dialog[open]")].length);
    console.log(theme, "dialogs open after Esc:", closed, "errors:", errors);
    await ctx.close();
  }
  await browser.close();
} finally {
  vite.kill();
}
