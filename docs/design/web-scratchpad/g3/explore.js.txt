const JUDGE_SCHEMA = { type: 'object', required: ['pick', 'why', 'loserGap', 'winnerFlaw'], properties: {
  pick: { type: 'string', enum: ['A', 'B', 'same'] },
  why: { type: 'string', description: 'The deciding differences, in two or three sentences' },
  loserGap: { type: 'string', description: "The losing one's single biggest gap to the winner, as one concrete instruction a designer can act on (if same: the bigger shared flaw)" },
  winnerFlaw: { type: 'string', description: "The winner's own single biggest remaining flaw, as one concrete instruction" },
  broken: { type: 'string', description: 'Anything broken in either: clipped or overlapping text, missing content or controls, broken layout, unreadable contrast (say which, A or B, and which screen)' },
} }

const files = (dir, tag, n) => Array.from({ length: n }, (_, i) => `${dir}\\${tag}-${i + 1}.png`)

function judgePrompt(dir, n, kind) {
  const pairs = Array.from({ length: n }, (_, i) => `Screen ${i + 1}: A ${dir}\\A-${i + 1}.png, B ${dir}\\B-${i + 1}.png`).join('\n')
  const what = kind === 'screens'
    ? `Two designs, A and B, of the same product's screens, with their labels stripped: the same content and state on each screen, designed two ways. They may differ in layout and in visual language; that's allowed. Judge which is the better-designed product UI for this content. If one leaves out content or controls the other shows, count it against it.`
    : `Two versions, A and B, of the same product screens, with their labels stripped: the same content in the same layout, drawn in two visual treatments. Where one shows a picture (a logo, an emoji, a photo) and the other a stand-in for it, don't count that either way.`
  return `You are a harsh product design critic. Praise is not useful.

${what}
${pairs}

Open every image and study A and B side by side, part by part, screen by screen; then weigh all the screens together. Judge the craft: visual hierarchy, typography, colour and contrast, density, spacing and alignment, separation and surfaces, glyphs, consistency, polish, and how clearly each screen does its job.

Answer A, B, or same. Say "same" only when you genuinely can't tell them apart in quality, never as a hedge between two that differ. Then name the loser's single biggest gap to the winner and the winner's own biggest remaining flaw, each as one concrete instruction, and list anything broken. Do not edit any file.`
}

// four judges; judges 1 and 2 see X (the direction) as A. Returns { X, Y, same, judges }
async function match(id, n, kind, label) {
  const runs = await parallel([1, 2, 3, 4].map((k) => () => {
    const xIs = k <= 2 ? 'A' : 'B'
    return agent(judgePrompt(`${X}judge\\${id}\\j${k}`, n, kind), { label: `${label}:j${k}`, phase: 'Judge', schema: JUDGE_SCHEMA, effort: 'high' })
      .then((v) => v && { judge: k, xIs, ...v, vote: v.pick === 'same' ? 'same' : (v.pick === xIs ? 'X' : 'Y') })
  }))
  const js = runs.filter(Boolean)
  return { X: js.filter((j) => j.vote === 'X').length, Y: js.filter((j) => j.vote === 'Y').length, same: js.filter((j) => j.vote === 'same').length, judges: js }
}

async function judge(d, b) {
  if (!b || !b.patch) return { key: d.key, dir: d.dir, built: false }
  const out = X + d.dir + '\\'
  const prep = await agent(`Run these two commands from ${ROOT} and report their output exactly. Do nothing else.
node ${G3}\\xprep.mjs ${X}judge ${d.dir}-s ${out}screens ${args.polish ? `${PREV}${d.dir}\\screens` : `${INC}screens`} screens
node ${G3}\\xprep.mjs ${X}judge ${d.dir}-t ${out}twins ${INC}twins linear`,
    { label: `prep:${d.key}`, phase: 'Judge', effort: 'medium' })
  if (!/-s: 10 per side/.test(prep) || !/-t: 7 per side/.test(prep)) return { key: d.key, dir: d.dir, built: true, prepFailed: prep, patch: b.patch, summary: b.summary }
  const [screens, twins] = await Promise.all([
    match(`${d.dir}-s`, 10, 'screens', `screens:${d.key}`),
    match(`${d.dir}-t`, 7, 'twins', `linear:${d.key}`),
  ])
  return { key: d.key, dir: d.dir, built: true, screens, twins, patch: b.patch, summary: b.summary, checks: b.checks, kit: b.kit }
}

phase('Build')

  (d, _, i) => agent(buildPrompt(d, 5241 + i), { label: `build:${d.key}`, phase: 'Build', schema: BUILD_SCHEMA, isolation: 'worktree', effort: args.effort }),
  (b, d) => judge(d, b),
)

const verdicts = results.filter(Boolean)
for (const r of verdicts) log(`${r.key}: screens ${r.screens ? `direction ${r.screens.X}, ${args.polish ? 'previous' : 'current'} ${r.screens.Y}, same ${r.screens.same}` : '-'}; twins vs Linear ${r.twins ? `direction ${r.twins.X}, Linear ${r.twins.Y}, same ${r.twins.same}` : '-'}`)
return { round: args.round, verdicts }
