const JUDGE_SCHEMA = { type: 'object', required: ['pick', 'why', 'loserGap', 'winnerFlaw'], properties: {
  pick: { type: 'string', enum: ['A', 'B', 'same'] },
  why: { type: 'string', description: 'The deciding differences, in two or three sentences' },
  loserGap: { type: 'string', description: "The losing one's single biggest gap to the winner, as one concrete instruction (if same: the bigger shared flaw)" },
  winnerFlaw: { type: 'string', description: "The winner's own single biggest remaining flaw, as one concrete instruction" },
  broken: { type: 'string', description: 'Anything broken in either: clipped or overlapping text, missing content, broken layout, unreadable contrast (say which, A or B)' },
} }

function judgePrompt(pairs, withRef) {
  const list = pairs.map((p, i) => `Screen ${i + 1}: A ${p.a}, B ${p.b}${withRef ? `, the standard ${p.r}` : ''}`).join('\n')
  const ref = withRef
    ? `\nEach screen also has "the standard": the same content in a reference product both are held to. It is not a contestant; judge A and B by how close they come to its craft, and where both fall short, which falls shorter.\n`
    : ''
  return `You are a harsh product design critic. Praise is not useful.

Two versions, A and B, of the same product screens, with their labels stripped: the same content, drawn two ways.
${list}
${ref}
Open every image and study A and B side by side, part by part, screen by screen; then weigh all the screens together. Which is the better-crafted product UI? Judge the craft: visual hierarchy, typography, colour and contrast, density, spacing and alignment, separation and surfaces, glyphs, consistency, polish. Where one shows a picture (a logo, an emoji, a photo) and the other a stand-in for it, don't count that either way.

Answer A, B, or same. Say "same" only when you genuinely can't tell them apart in quality, never as a hedge between two that differ. Then name the loser's single biggest gap to the winner and the winner's own biggest remaining flaw, each as one concrete instruction, and list anything broken. Do not edit any file.`
}

// four judges, two seeing the incumbent as A; returns { inc, cand, same, judges }
async function match(label, pairs, withRef) {
  const runs = await parallel([1, 2, 3, 4].map((k) => () => {
    const flip = k > 2
    const shown = pairs.map((p) => ({ a: flip ? p.cand : p.inc, b: flip ? p.inc : p.cand, r: p.ref }))
    return agent(judgePrompt(shown, withRef), { label: `${label}:j${k}`, phase: 'Judge', schema: JUDGE_SCHEMA })
      .then((v) => v && { judge: k, incIs: flip ? 'B' : 'A', ...v, vote: v.pick === 'same' ? 'same' : (v.pick === (flip ? 'B' : 'A') ? 'inc' : 'cand') })
  }))
  const js = runs.filter(Boolean)
  return { inc: js.filter((j) => j.vote === 'inc').length, cand: js.filter((j) => j.vote === 'cand').length, same: js.filter((j) => j.vote === 'same').length, judges: js }
}

async function judge(l, b) {
  if (!b || !b.patch) return { key: l.key, kept: false, reason: 'no challenger' }
  const out = K + l.dir + '\\'
  const twins = b.twins.map((t) => t.name.replace(/[\\/](twin|linear)\.png$/, '')).filter((n, i, a) => a.indexOf(n) === i)
  const screens = b.screens.slice().sort((x, y) => y.pixels - x.pixels).map((s) => s.path)
    .filter((p, i, a) => a.indexOf(p) === i).slice(0, MAX_SCREENS)
  if (b.screens.length > MAX_SCREENS) log(`${l.key}: ${b.screens.length} screens moved; the guard judges the ${MAX_SCREENS} that moved most`)
  const [tw, sc] = await Promise.all([
    twins.length
      ? match(`twins:${l.key}`, twins.map((n) => ({ inc: `${INC}twins\\${n}\\twin.png`, cand: `${out}twins\\${n}\\twin.png`, ref: `${INC}twins\\${n}\\linear.png` })), true)
      : null,
    screens.length
      ? match(`screens:${l.key}`, screens.map((p) => ({ inc: `${INC}screens\\${p}`, cand: `${out}screens\\${p}` })), false)
      : null,
  ])
  const kept = !!tw && tw.cand >= 3 && !(sc && sc.inc >= 3)
  return { key: l.key, kept, twins: tw, screens: sc, twinNames: twins, screenPaths: screens, patch: b.patch, summary: b.summary, checks: b.checks, kitLines: b.kitLines }
}
