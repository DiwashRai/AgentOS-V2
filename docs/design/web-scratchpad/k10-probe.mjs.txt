// One-off: a closed peek's toast doesn't greet the next peek; a picker's fact tip stays quiet after
// Esc; Menu's rest still holds. Vite on 5191, stopped at the end.
import { spawn } from "node:child_process";
import { createRequire } from "node:module";
import { join } from "node:path";

const [web] = process.argv.slice(2);
const { chromium } = createRequire(join(web, "package.json"))("@playwright/test");
const vite = spawn(process.execPath, ["node_modules/vite/bin/vite.js", "--port", "5191", "--strictPort"], { cwd: web, stdio: "ignore" });
const say = (k, v) => console.log(`${k}: ${typeof v === "string" ? v : JSON.stringify(v)}`);
try {
  const base = "http://127.0.0.1:5191";
  for (let i = 0; ; i++) {
    try { if ((await fetch(base)).ok) break; } catch {}
    if (i > 100) throw new Error("vite did not start");
    await new Promise((r) => setTimeout(r, 200));
  }
  const browser = await chromium.launch({ channel: "chrome" });
  const ctx = await browser.newContext({ viewport: { width: 1440, height: 900 }, permissions: ["clipboard-read", "clipboard-write"] });
  const page = await ctx.newPage();
  const errors = [];
  page.on("pageerror", (e) => errors.push(String(e)));
  const tip = () => page.evaluate(() => [...document.querySelectorAll("[class*=_tip_]")].filter((e) => e.checkVisibility()).map((e) => e.textContent));
  const toasts = () => page.evaluate(() => [...document.querySelectorAll("[class*=_toast_]")].map((t) => t.textContent));

  await page.goto(base + "/?peek=NW-17");
  await page.waitForTimeout(600);
  await page.locator("dialog").getByRole("button", { name: "More on NW-17" }).click();
  await page.getByRole("menuitem", { name: "Copy key" }).click();
  await page.waitForTimeout(400);
  say("peek NW-17 toasts", await toasts());
  await page.keyboard.press("Escape");
  await page.waitForTimeout(400);
  await page.getByText("Show the run's stage on the board").click();
  await page.waitForTimeout(600);
  say("next peek (NW-14) toasts", await toasts());
  await page.keyboard.press("Escape");
  await page.waitForTimeout(400);

  await page.goto(base + "/t/NW-17");
  await page.waitForTimeout(600);
  const owner = page.getByRole("combobox", { name: "Owner" });
  await owner.focus();
  await page.waitForTimeout(900);
  say("owner focused: tips", await tip());
  await page.keyboard.press("Enter");
  await page.waitForTimeout(300);
  await page.keyboard.press("Escape");
  await page.waitForTimeout(900);
  say("owner after Esc: focus, tips", [await page.evaluate(() => document.activeElement?.getAttribute("aria-label")), await tip()]);
  await page.keyboard.press("Tab");
  await page.keyboard.press("Shift+Tab");
  await page.waitForTimeout(900);
  say("owner left and back: tips", await tip());

  const more = page.getByRole("button", { name: "More on NW-17" });
  await more.focus();
  await page.keyboard.press("Enter");
  await page.waitForTimeout(300);
  await page.keyboard.press("Escape");
  await page.waitForTimeout(900);
  say("menu after Esc: tips", await tip());
  say("errors", errors);
  await browser.close();
} finally {
  vite.kill();
}
