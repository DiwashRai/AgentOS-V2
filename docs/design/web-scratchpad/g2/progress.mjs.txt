// Builds the gauntlet's progress page from the run's state file and the latest best shots.
// Usage: node progress.mjs <state.json> <out dir>  -> <out dir>/progress.html, <out dir>/files.json
// (files.json maps each published image path to its source, for the Artifact publish).
import { existsSync, readFileSync, readdirSync, writeFileSync } from "node:fs";
import { join } from "node:path";

const [statePath, outDir] = process.argv.slice(2);
const state = JSON.parse(readFileSync(statePath, "utf8"));
const G = "C:\\code\\AgentOS-V2-web\\.review\\gauntlet2";
const PIECES = {
  frame: "Frame and themes", lists: "My queue and Compact", intake: "Intake", teams: "All teams",
  ticket: "Ticket page", overlays: "Peek and editors over the ticket", library: "Library lists",
  editors: "Library editors", controls: "Controls and popovers",
};
const esc = (s) => String(s ?? "").replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" })[c]);

// the latest best shots of each piece: the newest round folder that has them
const files = {};
const latest = {};
const rounds = [0, ...state.rounds.map((r) => r.round)].sort((a, b) => b - a);
for (const key of Object.keys(PIECES)) {
  for (const r of rounds) {
    const dir = r === 0 ? join(G, "r0", key) : join(G, `r${r}`, "best", key);
    if (!existsSync(dir)) continue;
    const imgs = readdirSync(dir).filter((f) => f.endsWith(".png")).sort();
    if (!imgs.length) continue;
    latest[key] = { round: r, imgs: imgs.map((f) => { const pub = `shots/${key}/${f}`; files[pub] = join(dir, f); return pub; }) };
    break;
  }
}

const history = (key) => state.rounds.map((r) => ({ round: r.round, mode: r.mode, ...(r.verdicts.find((v) => v.key === key) || {}) }));
const keptBest = (h) => h.map((x) => (typeof x.cand === "number" ? Math.max(x.best, x.kept ? x.cand : x.best) : null));

function chart(key) {
  const h = history(key).filter((x) => typeof x.best === "number");
  if (!h.length) return `<p class="muted small">No rounds yet.</p>`;
  const W = 300, H = 120, P = 22;
  const xs = (i) => P + (h.length === 1 ? (W - 2 * P) / 2 : (i * (W - 2 * P)) / (h.length - 1));
  const ys = (v) => H - P - ((v - 0) / 10) * (H - 2 * P);
  const line = (vals, cls) => `<polyline class="${cls}" points="${vals.map((v, i) => `${xs(i)},${ys(v)}`).join(" ")}"/>` + vals.map((v, i) => `<circle class="${cls}" cx="${xs(i)}" cy="${ys(v)}" r="3"/>`).join("");
  const best = keptBest(h);
  return `<svg viewBox="0 0 ${W} ${H}" class="chart" role="img" aria-label="Scores by round">
    ${[0, 5, 10].map((v) => `<line class="grid" x1="${P}" x2="${W - P}" y1="${ys(v)}" y2="${ys(v)}"/><text class="tick" x="4" y="${ys(v) + 4}">${v}</text>`).join("")}
    ${line(h.map((x) => x.linear), "linear")}${line(best, "ours")}
    ${h.map((x, i) => `<text class="tick" x="${xs(i)}" y="${H - 4}" text-anchor="middle">r${x.round}</text>`).join("")}
  </svg>`;
}

function card(key) {
  const h = history(key);
  const last = h[h.length - 1] || {};
  const best = keptBest(h.filter((x) => typeof x.best === "number"));
  const now = best.length ? best[best.length - 1] : null;
  const recent = best.slice(-3);
  const flat = recent.length === 3 && Math.max(...recent) - Math.min(...recent) <= 0.5;
  const status = state.stopped?.includes(key) ? `<span class="pill stop">stopped: plateau</span>` : flat ? `<span class="pill stop">plateau</span>` : last.kept ? `<span class="pill win">challenger kept</span>` : h.length ? `<span class="pill">best held</span>` : `<span class="pill">baseline</span>`;
  const L = latest[key];
  return `<article class="piece" id="${key}">
    <header><h2>${esc(PIECES[key])}</h2>${status}</header>
    <div class="scores">
      <div><span class="big">${now ?? "–"}</span><span class="muted small">ours</span></div>
      <div><span class="big linear-t">${last.linear ?? "–"}</span><span class="muted small">Linear</span></div>
      ${chart(key)}
    </div>
    ${last.gap ? `<p class="gap"><b>Biggest gap:</b> ${esc(last.gap)}</p>` : ""}
    ${last.summary && last.kept ? `<p class="small muted"><b>Kept this round:</b> ${esc(last.summary)}</p>` : ""}
    <div class="shots">${(L?.imgs || []).map((src) => `<a href="${src}" target="_blank" rel="noopener"><img src="${src}" alt="${esc(PIECES[key])}: ${esc(src.split("/").pop())}" loading="lazy"></a>`).join("")}</div>
    <p class="small muted">Best shots from ${L ? (L.round === 0 ? "the baseline" : `round ${L.round}`) : "–"}.</p>
  </article>`;
}

<div class="wrap">
  <header class="top">
    <span class="eyebrow">AgentOS V2 web · branch web-restyle · Linear-style restyle</span>

    <p class="muted">Each round every piece gets one challenger beside its current best. A fresh critic scores Linear, our best and the challenger blind, out of 10, and the higher of ours is kept. A piece stops when its best stays within 0.5 for three rounds. ${esc(state.note || "")}</p>
    <div class="stats"><span><b>${doneRounds}</b> rounds done</span><span><b>${esc(state.status || "")}</b></span><span>last commit <code>${esc(state.rounds.at(-1)?.commit || state.base || "")}</code></span></div>
    <div class="legend"><span><i style="background:var(--ours)"></i>ours (best kept)</span><span><i style="background:var(--linear)"></i>Linear</span></div>
