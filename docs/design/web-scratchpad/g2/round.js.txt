  const dir = `${OUT}r${round}\\cand\\${p.key}`
  return `You build one challenger for one piece of a gauntlet that restyles the AgentOS V2 web app in Linear's design language: its look, not its features. Add no screen, feature or data the app doesn't have.

You are in your own git worktree (your current directory), a copy of ${ROOT}. Set it up first:
1. \`git rev-parse HEAD\` must be ${base}; if it isn't, run \`git checkout --detach ${base}\` (in this worktree only).
2. Link the installed packages, one junction per package, with the fonts copied for real (Vite refuses to serve files through a junction that leads outside the worktree, and the page then falls back to another font), in PowerShell from the worktree root:
   \`New-Item -ItemType Directory web\\node_modules | Out-Null; Get-ChildItem ${ROOT}\\web\\node_modules -Force | Where-Object { $_.Name -ne '.vite' } | ForEach-Object { if ($_.Name -eq '@fontsource') { Copy-Item $_.FullName web\\node_modules\\@fontsource -Recurse } else { New-Item -ItemType Junction -Path ('web\\node_modules\\' + $_.Name) -Target $_.FullName | Out-Null } }\`
   The camera fails if the Geist font didn't load: then fix the setup, never shoot without it.
Never commit, push, stash, or touch ${ROOT} itself except to write under ${OUT}.

Your piece: ${p.key}, ${spec.what}. Focus your edits on ${spec.focus}; touch other files only when your piece needs it.
${MODE[mode]}
${p.note ? `\nThe critic's notes on the current best are in ${p.note}: read them first.\n` : ''}
The reference: Linear's own screens, ${spec.refs.map((r) => REF + r).join(', ')}. Open and study them. The current best for your piece is ${p.best.join(', ')}: open those too.

The rules of the repo (AGENTS.md) hold:
- Only web/src/kit renders elements or holds CSS. Raw values live only in tokens.css; a new colour is a token, written from the base tokens (color-mix of --page, --ink-1, --you and so on) so every theme gets it.
- Every kit component, prop and variant has a reason in web/src/kit/ledger.json. Keep the entries you touch true.
- The budget is 45 components (44 used) and 9,000 lines (about 8,360 used), shared by 9 builders this round: add at most 60 kit lines net. Replace rather than add. Never change "budget" or "raises".
- Never weaken a check or a gate.

See your work: \`node ${SHOTS} <worktree>\\web ${port} ${dir} ${p.key}\` starts Vite on port ${port}, shoots your piece's states exactly as the critic will see them, and stops Vite. Look at every shot, compare with Linear's, and iterate as much as you need.

Before you finish, from web/: \`npx tsc -p . --noEmit\`, \`npx eslint src\`, \`node node_modules/stylelint/bin/stylelint.mjs "src/**/*.css"\`, \`npx prettier --write src\`, then \`node tools/kit.mjs --check\` (all must pass). Shoot your final state into ${dir}. Save your change as a patch: from the worktree root, \`git add -A -- web/src\` then \`git diff --cached --binary ${base} -- web/src > ${OUT}r${round}\\patches\\${p.key}.patch\` (create the folder). }

function shuffle(arr, salt) {
  const out = arr.slice()
  for (let i = out.length - 1; i > 0; i--) {
    const j = (i * 7 + salt * 13 + round * 5) % (i + 1)
    const t = out[i]; out[i] = out[j]; out[j] = t
  }
  return out
}

async function judge(p, spec, built, idx) {
  if (!built || !built.patch || !(built.shots || []).length) return { key: p.key, kept: false, reason: 'no challenger' }
  const sets = shuffle([
    { who: 'linear', images: spec.refs.map((r) => REF + r) },
    { who: 'best', images: p.best },
    { who: 'cand', images: built.shots },
  ], idx)
  const names = ['X', 'Y', 'Z']
  sets.forEach((s, i) => { s.name = names[i] })
  const v = await agent(`You are a harsh design critic. Praise is not useful. Below are three sets of product screenshots, labels stripped. Some may come from different products, with different content and framing (some are close crops); judge the design, not the content or the crop: visual hierarchy, typography, colour and contrast, density, separation, alignment, consistency, polish and craft, as a modern product UI.

${sets.map((s) => `Set ${s.name}: ${s.images.join(', ')}`).join('\n')}

Open every image. Score each set out of 10, to one decimal, on the same scale, as strictly as you would score a shipped product. Two sets may look alike: score each on its own merits, and don't split the difference. For each set name its single biggest design gap, as one concrete instruction a designer can act on, and list anything broken (clipped or overlapping text, missing content, broken layout, unreadable contrast). Do not edit any file.`,
    { label: `judge:${p.key}:r${round}`, phase: 'Judge', schema: VERDICT_SCHEMA })
  if (!v) return { key: p.key, kept: false, reason: 'critic failed' }
  const score = (who) => v.sets.find((x) => x.set.includes(sets.find((s) => s.who === who).name)) || {}
  const L = score('linear'), B = score('best'), C = score('cand')
  const kept = typeof C.score === 'number' && typeof B.score === 'number' && C.score > B.score
  return {
    key: p.key, kept, linear: L.score, best: B.score, cand: C.score,
    gap: (kept ? C : B).gap, problems: (kept ? C : B).problems || '', linearGap: L.gap,
    patch: built.patch, shots: built.shots, summary: built.summary, checks: built.checks,
  }
}

phase('Build')
const results = await pipeline(
  args.pieces,
  (p, _, i) => agent(buildPrompt(p, PIECES[p.key], 5241 + i), { label: `build:${p.key}:r${round}`, phase: 'Build', schema: BUILD_SCHEMA, isolation: 'worktree' }),
  (built, p, i) => judge(p, PIECES[p.key], built, i),
)

const verdicts = results.filter(Boolean)
for (const r of verdicts) log(`${r.key}: Linear ${r.linear}, best ${r.best}, challenger ${r.cand}${r.kept ? ' KEPT' : ''}`)
const winners = verdicts.filter((r) => r.kept).sort((a, b) => (b.cand - b.best) - (a.cand - a.best))

phase('Integrate')
const active = args.pieces.map((p) => p.key)
const integ = await agent(`You integrate one round of a gauntlet restyling the AgentOS V2 web app, in ${ROOT} (branch web-restyle, at ${base}). Do not push, switch branches or stash.

Apply these winning patches, in this order (the biggest win first), each with \`git apply --3way <patch>\` from the repo root:
${winners.map((w) => `- ${w.key}: ${w.patch} (${w.summary.replace(/\n/g, ' ')})`).join('\n') || '(none: nothing won this round)'}

Where two patches touch the same thing (tokens.css, a shared component, the ledger), resolve it so both designs hold. If a patch can't be made to fit, drop it, say which and why. Keep the screens' content and behaviour.

Then make the tree sound, never by weakening a check: from the repo root, \`just check\`, \`just test\` and \`just gates\` all pass. The kit must stay within its budget (45 components, 9,000 lines): if the kept patches go over, trim the least important additions, and say what you trimmed. Make every ledger reason true to the code.

Then \`just shots --update-snapshots\` (the screens changed on purpose; look at a few of the new baselines to be sure none is broken), and commit everything with a message file (\`git commit -F <file>\`): a subject "gauntlet: round ${round}", a line per kept piece, and the trailer "Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>". The pre-commit hook runs the checks: if it fails, fix and commit again. Never use --no-verify.

Finally shoot the new best for every piece: \`node ${SHOTS} ${ROOT}\\web 5250 ${OUT}r${round}\\best ${active.join(',')}\`. Return the commit, what was applied or dropped, the checks, and the kit's numbers.`,
  { label: `integrate:r${round}`, phase: 'Integrate', schema: { type: 'object', required: ['commit', 'applied', 'dropped', 'checks'], properties: {
    commit: { type: 'string' }, applied: { type: 'array', items: { type: 'string' } }, dropped: { type: 'string' },
    checks: { type: 'string' }, kit: { type: 'string' }, notes: { type: 'string' },
  } } })

return { round, mode, verdicts, integration: integ }
