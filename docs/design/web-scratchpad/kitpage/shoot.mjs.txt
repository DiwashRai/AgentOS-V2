// One-off: screenshot each section of the /kit sheet in both themes, for the kit catalogue page.
// Usage: node shoot.mjs <web dir> <out dir>. Starts its own Vite on 5191 and stops it.
import { spawn } from "node:child_process";
import { mkdirSync } from "node:fs";
import { createRequire } from "node:module";
import { join } from "node:path";

const [web, out] = process.argv.slice(2);
mkdirSync(out, { recursive: true });
const require = createRequire(join(web, "package.json"));
const { chromium } = require("@playwright/test");
const PORT = 5191;
const vite = spawn(process.execPath, ["node_modules/vite/bin/vite.js", "--port", String(PORT), "--strictPort"], {
  cwd: web,
  stdio: "ignore",
});
try {
  const base = `http://127.0.0.1:${PORT}`;
  for (let i = 0; ; i++) {
    try {
      if ((await fetch(base)).ok) break;
    } catch {}
    if (i > 100) throw new Error("vite did not start");
    await new Promise((r) => setTimeout(r, 200));
  }
  const browser = await chromium.launch({ channel: "chrome" });
  for (const theme of ["light", "dark"]) {
    const ctx = await browser.newContext({
      viewport: { width: 1100, height: 900 },
      deviceScaleFactor: 2,
      timezoneId: "Europe/London",
      locale: "en-GB",
    });
    await ctx.addInitScript((t) => localStorage.setItem("agentos-theme", t), theme);
    const page = await ctx.newPage();
    await page.goto(`${base}/kit`);
    await page.evaluate(() => document.fonts.ready);
    const n = await page.evaluate(() => {
      const leaf = (t) =>
        [...document.querySelectorAll("body *")].find((e) => e.children.length === 0 && e.textContent === t);
      const a = leaf("Text");
      const b = leaf("Controls");
      let p = a;
      while (!p.contains(b)) p = p.parentElement;
      [...p.children].forEach((c, i) => c.setAttribute("data-sec", String(i)));
      return p.children.length;
    });
    for (let i = 0; i < n; i++)
      await page.locator(`[data-sec="${i}"]`).screenshot({ path: join(out, `sec${i}-${theme}.png`), animations: "disabled" });
    await ctx.close();
  }
  await browser.close();
} finally {
  vite.kill();
}
