// One-off: toasts raised from effects at the component seam. A full Dialog rendered with the app's
// own React: a child that raises on mount, and one that raises as the question opens; plus a
// callback taken before the question. Vite on 5191, stopped at the end.
import { spawn } from "node:child_process";
import { createRequire } from "node:module";
import { join } from "node:path";

const [web] = process.argv.slice(2);
const { chromium } = createRequire(join(web, "package.json"))("@playwright/test");
const vite = spawn(process.execPath, ["node_modules/vite/bin/vite.js", "--port", "5191", "--strictPort"], { cwd: web, stdio: "ignore" });
const say = (k, v) => console.log(`${k}: ${typeof v === "string" ? v : JSON.stringify(v)}`);
try {
  const base = "http://127.0.0.1:5191";
  for (let i = 0; ; i++) {
    try { if ((await fetch(base)).ok) break; } catch {}
    if (i > 100) throw new Error("vite did not start");
    await new Promise((r) => setTimeout(r, 200));
  }
  const browser = await chromium.launch({ channel: "chrome" });
  const page = await browser.newPage({ viewport: { width: 1440, height: 900 } });
  const errors = [];
  page.on("pageerror", (e) => errors.push(String(e)));
  await page.clock.install();
  await page.goto(base + "/kit");
  await page.waitForTimeout(600);
  await page.evaluate(async () => {
    const dep = async (from, name) => {
      const src = await (await fetch(from)).text();
      const m = src.match(new RegExp(`from "(/node_modules/\\.vite/deps/${name}\\.js\\?v=[^"]+)"`));
      return import(m[1]);
    };
    const R = await dep("/src/kit/Dialog.tsx", "react");
    const React = R.createElement ? R : R.default;
    const D = await dep("/src/main.tsx", "react-dom_client");
    const createRoot = D.createRoot ?? D.default.createRoot;
    const kit = await import("/src/kit/index.ts");
    const h = React.createElement;
    function Mounting() {
      const raise = kit.useToast();
      React.useEffect(() => { raise("Mount result"); }, []);
      return h("p", null, "mounting");
    }
    function Asked({ asking }) {
      const raise = kit.useToast();
      window.__early = window.__early ?? raise;
      React.useEffect(() => { if (asking) raise("Effect held result"); }, [asking, raise]);
      return h("p", null, "asked");
    }
    function Probe() {
      const [asking, setAsking] = React.useState(false);
      window.__ask = () => setAsking(true);
      window.__answer = () => setAsking(false);
      return h(kit.Dialog, {
        label: "Probe", open: true, onClose: () => undefined, look: "full",
        ask: asking ? { question: "Save?", actions: h(kit.Button, { onClick: () => setAsking(false) }, "Keep editing") } : undefined,
      }, h("div", null, h(Mounting), h(Asked, { asking })));
    }
    const div = document.createElement("div");
    document.body.append(div);
    createRoot(div).render(h(Probe));
  });
  const shown = () => page.evaluate(() => [...document.querySelectorAll("dialog[open] [class*=_toast_]")].map((t) => t.textContent));
  const ax = async (t) => (await page.locator("body").ariaSnapshot()).includes(t);
  await page.waitForTimeout(500);
  say("1 mount effect: shown", await shown());
  await page.clock.runFor(8000);
  await page.waitForTimeout(300);
  await page.evaluate(() => window.__ask());
  await page.waitForTimeout(300);
  await page.evaluate(() => window.__early("Early callback result"));
  await page.waitForTimeout(300);
  say("2 while asking: shown", await shown());
  await page.clock.runFor(8000);
  await page.evaluate(() => window.__answer());
  await page.waitForTimeout(500);
  say("2 answered: shown, in AX", [await shown(), await ax("Effect held result"), await ax("Early callback result")]);
  say("errors", errors);
  await browser.close();
} finally {
  vite.kill();
}
